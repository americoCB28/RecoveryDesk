cmake_minimum_required(VERSION 3.24)
project(RecoveryDesk VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "RecoveryDesk requires a Windows x64 toolchain")
endif()
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Only x64 builds are supported")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(MSVC)
  set(RD_PE_TOOLCHAIN true)
else()
  set(RD_PE_TOOLCHAIN false)
endif()
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/build-info.json"
  "{\n  \"app_version\": \"${PROJECT_VERSION}\",\n  \"architecture\": \"x64\",\n  \"compiler\": \"${CMAKE_CXX_COMPILER_ID} ${CMAKE_CXX_COMPILER_VERSION}\",\n  \"winpe_builder_toolchain_supported\": ${RD_PE_TOOLCHAIN},\n  \"hardware_certified\": false\n}\n")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
if(MSVC)
  add_compile_options(/W4 /permissive- /utf-8 /EHsc /guard:cf)
  add_link_options(/DYNAMICBASE /NXCOMPAT /guard:cf)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -finput-charset=UTF-8)
  add_link_options(-static -municode)
endif()
add_library(recoverydesk_core STATIC src/core.cpp src/windows.cpp src/backup.cpp src/media.cpp src/boot_probe.cpp)
target_include_directories(recoverydesk_core PUBLIC include PRIVATE third_party)
target_link_libraries(recoverydesk_core PUBLIC bcrypt ole32 oleaut32 wbemuuid advapi32 shell32)
add_executable(RecoveryDesk WIN32 apps/main.cpp apps/app.rc)
target_include_directories(RecoveryDesk PRIVATE third_party apps)
target_link_libraries(RecoveryDesk PRIVATE recoverydesk_core comctl32 comdlg32 gdi32 uxtheme uuid)
add_custom_command(TARGET RecoveryDesk POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/data" "$<TARGET_FILE_DIR:RecoveryDesk>/data"
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/boot" "$<TARGET_FILE_DIR:RecoveryDesk>/boot"
  COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_SOURCE_DIR}/LICENSE" "$<TARGET_FILE_DIR:RecoveryDesk>/LICENSE"
  COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_SOURCE_DIR}/NOTICE" "$<TARGET_FILE_DIR:RecoveryDesk>/NOTICE"
  COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_CURRENT_BINARY_DIR}/build-info.json" "$<TARGET_FILE_DIR:RecoveryDesk>/build-info.json"
  COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:RecoveryDesk>/third_party/nlohmann"
  COMMAND ${CMAKE_COMMAND} -E copy "${CMAKE_SOURCE_DIR}/third_party/nlohmann/LICENSE.MIT" "$<TARGET_FILE_DIR:RecoveryDesk>/third_party/nlohmann/LICENSE.MIT")
include(CTest)
if(BUILD_TESTING)
  add_executable(recoverydesk_tests tests/core_tests.cpp)
  target_include_directories(recoverydesk_tests PRIVATE third_party)
  target_link_libraries(recoverydesk_tests PRIVATE recoverydesk_core)
  add_test(NAME core_and_safety COMMAND recoverydesk_tests "${CMAKE_SOURCE_DIR}/data/guides.es.json")
endif()
install(TARGETS RecoveryDesk RUNTIME DESTINATION .)
install(DIRECTORY data boot DESTINATION .)
install(FILES LICENSE NOTICE README.md CONTRIBUTING.md SECURITY.md "${CMAKE_CURRENT_BINARY_DIR}/build-info.json" DESTINATION .)
install(DIRECTORY docs DESTINATION .)
install(FILES third_party/nlohmann/LICENSE.MIT DESTINATION third_party/nlohmann)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "RecoveryDesk-${PROJECT_VERSION}-windows-x64")
include(CPack)
